// WEBNest Kinetic Motion Engine: Synchronizing Lenis 120fps physics with GSAP ScrollTrigger
import Lenis from '@studio-freight/lenis';
import { gsap } from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';

export class WEBNestMotionEngine {
  private lenis: Lenis;

  constructor() {
    this.lenis = new Lenis({
      lerp: 0.08,            // Fluid linear interpolation (zero jitter)
      smoothWheel: true,      // Trackpad & mousewheel normalization
      wheelMultiplier: 0.88,
    });
  }

  public init(): void {
    gsap.registerPlugin(ScrollTrigger);

    // Direct coupling between Lenis scroll events and ScrollTrigger calculations
    this.lenis.on('scroll', ScrollTrigger.update);

    // Synchronize physics ticker on GSAP's high-precision 120fps loop
    gsap.ticker.add((time: number) => {
      this.lenis.raf(time * 1000);
    });

    // Zero lag smoothing guarantees instant rubberband response
    gsap.ticker.lagSmoothing(0);
  }
}
// WEBNest Native Android: Reactive Coroutine Stream with Jetpack Compose
package studio.webnest.android.ui

import androidx.compose.runtime.*
import androidx.compose.animation.*
import kotlinx.coroutines.flow.StateFlow

@Composable
fun WebNestAppScreen(viewModel: StudioViewModel = hiltViewModel()) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    Scaffold(
        topBar = { AnimatedTopBar(title = "WEBNest Studio") }
    ) { paddingValues ->
        Crossfade(
            targetState = uiState.connectionStatus,
            animationSpec = spring(dampingRatio = Spring.DampingRatioMediumBouncy)
        ) { status ->
            when (status) {
                Status.Ready   -> ClientPortalDashboard(uiState.projects)
                Status.Syncing -> EdgeNetworkSyncIndicator(latencyMs = 38)
            }
        }
    }
}
